"""
server.py
This module starts the web server and handles incoming requests.
"""
from flask import Flask, render_template, request
from EmotionDetection.emotion_detection import emotion_detector

app = Flask("Emotion Detector")

@app.route("/emotionDetector")
def detector():
    """ Handles Emotion Detection """
    text_to_analyze = request.args.get("textToAnalyze")
    res = emotion_detector(text_to_analyze)
    if res['dominant_emotion'] is None:
        response ="Invalid text! Please try again!."
    else:
        response = (f'For the given statement, the system response is "anger":{ res["anger"]},'
                f'"disgust": {res["disgust"]}, "fear": {res["fear"]}, "joy": {res["joy"]}'
                f'and "sadness": {res["sadness"]}.'
                f'The dominant emotion is {res["dominant_emotion"]}.')
    return response

@app.route("/")
def render_index():
    """ Renders the index page """
    return render_template('index.html')
if __name__ == "__main__":
    app.run(host="0.0.0.0",port=5000)
