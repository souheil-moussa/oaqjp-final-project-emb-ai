import requests
import json
def emotion_detector(text_to_analyze):
    url='https://sn-watson-emotion.labs.skills.network/v1/watson.runtime.nlp.v1/NlpService/EmotionPredict'
    headers={"grpc-metadata-mm-model-id": "emotion_aggregated-workflow_lang_en_stock"}
    input_json={"raw_document": {"text":text_to_analyze}}
    res = requests.post(url,json=input_json,headers=headers)
    if res.status_code == 200:
        f_res = json.loads(res.text)
        dom='anger'
        anger_score = float(f_res["emotionPredictions"][0]["emotion"]["anger"])
        dom_value=anger_score
        disgust_score = float(f_res["emotionPredictions"][0]["emotion"]["disgust"])
        if disgust_score > dom_value:
            dom_value=disgust_score
            dom="disgust"
        joy_score = float(f_res["emotionPredictions"][0]["emotion"]["joy"])
        if joy_score > dom_value:
            dom_value=joy_score
            dom="joy"
        fear_score = float(f_res["emotionPredictions"][0]["emotion"]["fear"])
        if fear_score > dom_value:
            dom_value=fear_score
            dom="fear"
        sadness_score = float(f_res["emotionPredictions"][0]["emotion"]["sadness"])
        if sadness_score > dom_value:
            dom_value=sadness_score
            dom="sadness"
        out_dic={
        'anger': anger_score,
        'disgust': disgust_score,
        'fear': fear_score,
        'joy': joy_score,
        'sadness': sadness_score,
        'dominant_emotion': dom
        }
    elif res.status_code == 400:
        out_dic={
        'anger': None,
        'disgust': None,
        'fear': None,
        'joy': None,
        'sadness': None,
        'dominant_emotion': None
        }

    return out_dic
